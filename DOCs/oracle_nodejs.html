Instalación y Configuración del Entorno

Antes de empezar, verificar que Oracle XE está corriendo: Get-Service "OracleServiceXE" debe mostrar Running.

Requisitos previos

  • Oracle XE 21c instalado y corriendo (ver oracle.html sección 2)
  • PDB XEPDB1 abierta en modo READ WRITE (ver oracle.html sección 4)
  • Usuario admin_web creado en XEPDB1 (ver oracle.html sección 5)
  • Tablas del proyecto creadas (ver oracle.html sección 7)
  • Node.js v18+ instalado

npm install oracledb   # Driver Oracle para Node.js (v6+ soporta modo Thin)

Conexión Básica a Oracle XE con oracledb (Modo Thin)

Antes de construir el servidor completo, conviene probar la conexión con un script simple.

db/connection.js — Módulo de conexión simple
const oracledb = require('oracledb');

// MODO THIN: no requiere Oracle Client adicional
// Compatible con Oracle XE 21c y la PDB XEPDB1
oracledb.thin = true;   // Activar explícitamente modo Thin

const configuracionOracle = {
    user:          'admin_web',
    password:      'AdminWeb2024#',
    connectString: 'localhost:1521/XEPDB1'   // PDB de Oracle XE 21c
};

// Abrir una conexión individual (para pruebas o transacciones manuales)
async function obtenerConexion() {
    try {
        const conexion = await oracledb.getConnection(configuracionOracle);
        console.log('Conexión a Oracle XE (XEPDB1) exitosa');
        return conexion;
    } catch (errorDeConexion) {
        console.error('Error al conectar a Oracle XE:', errorDeConexion.message);
        throw errorDeConexion;
    }
}

// Cerrar la conexión de forma segura (usar en bloque finally)
async function cerrarConexion(conexion) {
    if (conexion) {
        try {
            await conexion.close();
        } catch (errorAlCerrar) {
            console.error('Error al cerrar conexión:', errorAlCerrar.message);
        }
    }
}

module.exports = { obtenerConexion, cerrarConexion };

const { obtenerConexion, cerrarConexion } = require('./db/connection');

        // PASO 2: Verificar que estamos conectados a la PDB correcta
        const resultadoPDB = await conexion.execute(
            "SELECT SYS_CONTEXT('USERENV','CON_NAME') AS pdb FROM dual"
        );
        console.log('PDB activa:', resultadoPDB.rows[0][0]);

        // PASO 3: Contar tablas del esquema admin_web
        const resultadoTablas = await conexion.execute(
            "SELECT COUNT(*) AS total FROM user_tables"

        console.error('Error:', error.message);
        console.error('Verifique: 1) OracleServiceXE activo  2) XEPDB1 abierta  3) usuario admin_web existe');
    } finally {
        // El bloque finally garantiza que la conexión SIEMPRE se cierra
        await cerrarConexion(conexion);
    }
}

Pool de Conexiones — Conexión Eficiente

El pool mantiene conexiones pre-abiertas para reutilizarlas. Evita la sobrecarga de abrir/cerrar una conexión en cada petición HTTP, lo que es crítico en una API REST.

Para Oracle XE 21c, poolMax no debe superar ~10 por los límites de la edición (2 CPUs, 2 GB RAM).
db/pool.js — Pool de conexiones Oracle XE
const oracledb = require('oracledb');

oracledb.thin = true;   // Modo Thin — sin Oracle Client adicional

// Configuración del pool usando variables de entorno (ver sección 9 — .env)
const configuracionPool = {
    user:          process.env.DB_USER     || 'admin_web',
    password:      process.env.DB_PASSWORD || 'AdminWeb2024#',
    connectString: process.env.DB_CONNECT  || 'localhost:1521/XEPDB1',

    // Ajuste para Oracle XE 21c (límite de 2 CPUs y 2 GB RAM):
    poolMin:       1,    // Mínimo de conexiones siempre abiertas
    poolMax:       5,    // Máximo simultáneo (XE: no superar 10)
    poolIncrement: 1,    // Conexiones adicionales por demanda
    poolTimeout:   60,   // Segundos de inactividad antes de cerrar

    // Retornar filas como objetos { COLUMNA: valor } en vez de arrays
    outFormat: oracledb.OUT_FORMAT_OBJECT
};

// Crear el pool al iniciar el servidor (llamar una sola vez)
async function iniciarPool() {
    try {
        await oracledb.createPool(configuracionPool);
        console.log(`Pool Oracle XE iniciado: ${configuracionPool.connectString}`);
    } catch (errorDePool) {
        console.error('Error al crear pool Oracle XE:', errorDePool.message);
        throw errorDePool;
    }
}

// Cerrar el pool al apagar el servidor
async function cerrarPool() {
    try {
        await oracledb.getPool().close(10);  // Espera 10s para que terminen las queries
        console.log('Pool Oracle XE cerrado correctamente');
    } catch (errorAlCerrar) {
        console.error('Error al cerrar pool:', errorAlCerrar.message);
    }
}

// Función base para ejecutar cualquier query usando el pool
async function ejecutarQuery(consultaSQL, parametros = [], opcionesExtra = {}) {
    let conexionDelPool;
    try {
        conexionDelPool = await oracledb.getConnection();   // Tomar del pool
        const resultadoConsulta = await conexionDelPool.execute(consultaSQL, parametros, {
            outFormat:  oracledb.OUT_FORMAT_OBJECT,
            autoCommit: true,
            ...opcionesExtra
        });
        return resultadoConsulta;
    } catch (errorDeQuery) {
        console.error('Error en query Oracle XE:', errorDeQuery.message);
        throw errorDeQuery;
    } finally {
        // Siempre devolver la conexión al pool (incluso si hubo error)
        if (conexionDelPool) await conexionDelPool.close();
    }
}

module.exports = { iniciarPool, cerrarPool, ejecutarQuery };

Módulo de Base de Datos Reutilizable

Este módulo encapsula las operaciones más comunes sobre el pool para que los modelos no repitan lógica de conexión.

db/db.js — Métodos CRUD genéricos sobre Oracle XE
const { ejecutarQuery } = require('./pool');
const oracledb = require('oracledb');

const db = {

    // SELECT — Devuelve un array de objetos [{ COL: valor }, ...]
    async query(consultaSQL, parametros = []) {
        const resultado = await ejecutarQuery(consultaSQL, parametros);
        return resultado.rows;   // Devuelve el arreglo de filas
    },

    // SELECT — Devuelve solo el primer objeto, o null si no existe
    async queryOne(consultaSQL, parametros = []) {
        const resultado = await ejecutarQuery(consultaSQL, parametros);
        return resultado.rows[0] || null;   // null si no encontró
    },

    // INSERT / UPDATE / DELETE — Devuelve cantidad de filas afectadas
    async execute(consultaSQL, parametros = []) {
        const resultado = await ejecutarQuery(consultaSQL, parametros, { autoCommit: true });
        return {
            filasAfectadas: resultado.rowsAffected,
            datosRetornados: resultado.outBinds
        };
    },

    // INSERT con retorno del ID generado (usando RETURNING INTO)
    async insertarConId(consultaSQL, parametros = {}) {
        const resultado = await ejecutarQuery(consultaSQL, parametros, {
            autoCommit: true,
            outFormat:  oracledb.OUT_FORMAT_OBJECT
        });
        return resultado.outBinds;
    }
};

module.exports = db;

JavaScript — Modelo de usuarios con operaciones CRUD
const db = require('../db/db');

const UsuarioModel = {

    // Listar todos los usuarios con su rol asignado
    async listarTodos() {
        const consultaListadoCompleto = `
            SELECT u.USUARIO_ID,
                   u.NOMBRE || ' ' || u.APELLIDO AS NOMBRE_COMPLETO,
                   u.EMAIL, u.USERNAME,
                   r.NOMBRE AS ROL, u.ACTIVO,
                   TO_CHAR(u.FECHA_REGISTRO, 'DD/MM/YYYY') AS FECHA_REGISTRO
            FROM   usuarios u
            JOIN   roles r ON u.rol_id = r.rol_id
            ORDER  BY u.nombre
        `;
        return await db.query(consultaListadoCompleto);
    },

    // Buscar un usuario por su ID numérico
    async buscarPorId(idDeUsuario) {
        const consultaPorId = `
            SELECT u.*, r.nombre AS nombre_rol
            FROM   usuarios u JOIN roles r ON u.rol_id = r.rol_id
            WHERE  u.usuario_id = :idDeUsuario
        `;
        return await db.queryOne(consultaPorId, [idDeUsuario]);
    },

    // Buscar usuario por email (para login)
    async buscarPorEmail(emailBuscado) {
        const consultaPorEmail = `
            SELECT u.*, r.nombre AS nombre_rol
            FROM   usuarios u JOIN roles r ON u.rol_id = r.rol_id
            WHERE  LOWER(u.email) = LOWER(:emailBuscado) AND u.activo = 'S'
        `;
        return await db.queryOne(consultaPorEmail, [emailBuscado]);
    },

    // Insertar un nuevo usuario (los parámetros nombrados evitan SQL Injection)
    async crear(datosDelNuevoUsuario) {
        const consultaInsercion = `
            INSERT INTO usuarios
                (rol_id, nombre, apellido, email, username, password_hash, telefono)
            VALUES
                (:rol_id, :nombre, :apellido, :email, :username, :password_hash, :telefono)
        `;
        return await db.execute(consultaInsercion, {
            rol_id:        datosDelNuevoUsuario.rol_id,
            nombre:        datosDelNuevoUsuario.nombre,
            apellido:      datosDelNuevoUsuario.apellido,
            email:         datosDelNuevoUsuario.email,
            username:      datosDelNuevoUsuario.username,
            password_hash: datosDelNuevoUsuario.password_hash,
            telefono:      datosDelNuevoUsuario.telefono || null
        });
    },

    // Actualizar datos de un usuario existente
    async actualizar(idDeUsuario, datosActualizados) {
        const consultaActualizacion = `
            UPDATE usuarios
            SET    nombre   = :nombre,
                   apellido = :apellido,
                   telefono = :telefono,
                   activo   = :activo
            WHERE  usuario_id = :idDeUsuario
        `;
        return await db.execute(consultaActualizacion, {
            nombre:      datosActualizados.nombre,
            apellido:    datosActualizados.apellido,
            telefono:    datosActualizados.telefono || null,
            activo:      datosActualizados.activo || 'S',
            idDeUsuario: idDeUsuario
        });
    },

    // Baja lógica: marcar como inactivo en vez de borrar físicamente
    async eliminar(idDeUsuario) {
        return await db.execute(
            "UPDATE usuarios SET activo = 'N' WHERE usuario_id = :idDeUsuario",
            [idDeUsuario]
        );
    }
};

module.exports = UsuarioModel;

JavaScript — Modelo de productos
const db = require('../db/db');

const ProductoModel = {

    // Listar productos activos con nombre de categoría
    async listarActivos() {
        const consultaProductosActivos = `
            SELECT p.PRODUCTO_ID, c.NOMBRE AS CATEGORIA,
                   p.NOMBRE, p.PRECIO, p.STOCK, p.IMAGEN_URL
            FROM   productos p
            JOIN   categorias c ON p.categoria_id = c.categoria_id
            WHERE  p.activo = 'S'
            ORDER  BY c.nombre, p.nombre
        `;
        return await db.query(consultaProductosActivos);
    },

    // Buscar producto por nombre (búsqueda parcial, insensible a mayúsculas)
    async buscarPorNombre(terminoDeBusqueda) {
        return await db.query(`
            SELECT p.PRODUCTO_ID, p.NOMBRE, p.PRECIO, p.STOCK, c.NOMBRE AS CATEGORIA
            FROM   productos p JOIN categorias c ON p.categoria_id = c.categoria_id
            WHERE  UPPER(p.nombre) LIKE UPPER('%' || :termino || '%') AND p.activo = 'S'
            ORDER  BY p.nombre
        `, [terminoDeBusqueda]);
    },

    // Crear producto nuevo en la base de datos
    async crear(datosDelProducto) {
        return await db.execute(`
            INSERT INTO productos (categoria_id, nombre, descripcion, precio, stock, imagen_url)
            VALUES (:categoria_id, :nombre, :descripcion, :precio, :stock, :imagen_url)
        `, {
            categoria_id: datosDelProducto.categoria_id,
            nombre:       datosDelProducto.nombre,
            descripcion:  datosDelProducto.descripcion || null,
            precio:       datosDelProducto.precio,
            stock:        datosDelProducto.stock || 0,
            imagen_url:   datosDelProducto.imagen_url || null
        });
    },

    // Baja lógica del producto
    async eliminar(idDelProducto) {
        return await db.execute(
            "UPDATE productos SET activo = 'N' WHERE producto_id = :id",
            [idDelProducto]
        );
    }
};

models/pedidoModel.js — Transacciones manuales

JavaScript — Crear pedido con transacción (rollback en caso de error)
const db       = require('../db/db');
const oracledb  = require('oracledb');

const PedidoModel = {

    // Crear un pedido completo con varios items en una sola transacción
    async crear(idDeUsuario, listaDeItems, direccionDeEnvio) {
        let conexionTransaccional;
        try {
            conexionTransaccional = await oracledb.getConnection();

            // PASO 1: Insertar el encabezado del pedido
            await conexionTransaccional.execute(
                `INSERT INTO pedidos (usuario_id, estado, direccion_envio)
                 VALUES (:uid, 'PENDIENTE', :dir)`,
                { uid: idDeUsuario, dir: direccionDeEnvio },
                { autoCommit: false }   // NO confirmar aún — parte de la transacción
            );

            // PASO 2: Obtener el pedido_id generado automáticamente
            const resultadoId = await conexionTransaccional.execute(
                `SELECT MAX(pedido_id) AS pid FROM pedidos WHERE usuario_id = :uid`,
                [idDeUsuario],
                { outFormat: oracledb.OUT_FORMAT_OBJECT }
            );
            const idDelNuevoPedido = resultadoId.rows[0].PID;

            // PASO 3: Insertar cada item — el trigger actualiza el total automáticamente
            for (const itemDelCarrito of listaDeItems) {
                await conexionTransaccional.execute(
                    `INSERT INTO detalle_pedidos
                         (pedido_id, producto_id, cantidad, precio_unitario)
                     VALUES (:pid, :prod, :cant, :precio)`,
                    {
                        pid:    idDelNuevoPedido,
                        prod:   itemDelCarrito.producto_id,
                        cant:   itemDelCarrito.cantidad,
                        precio: itemDelCarrito.precio_unitario
                    },
                    { autoCommit: false }
                );

                // PASO 4: Descontar stock (trg_validar_stock lanza error si queda negativo)
                await conexionTransaccional.execute(
                    `UPDATE productos SET stock = stock - :cant
                     WHERE producto_id = :pid AND stock >= :cant`,
                    { cant: itemDelCarrito.cantidad, pid: itemDelCarrito.producto_id },
                    { autoCommit: false }
                );
            }

            // PASO 5: Confirmar toda la transacción si no hubo errores
            await conexionTransaccional.commit();
            return { pedido_id: idDelNuevoPedido, items: listaDeItems.length };

        } catch (errorDeLaTransaccion) {
            // Si algo falló, deshacer todos los cambios
            if (conexionTransaccional) await conexionTransaccional.rollback();
            throw errorDeLaTransaccion;
        } finally {
            if (conexionTransaccional) await conexionTransaccional.close();
        }
    }
};

module.exports = PedidoModel;

API REST con Express + Oracle XE

routes/usuarios.js

JavaScript — Rutas CRUD de usuarios
const express      = require('express');
const enrutadorUsuarios = express.Router();
const UsuarioModel = require('../models/usuarioModel');

// GET /api/usuarios — Listar todos
enrutadorUsuarios.get('/', async (peticion, respuesta) => {
    try {
        const listaDeUsuarios = await UsuarioModel.listarTodos();
        respuesta.json({ ok: true, data: listaDeUsuarios, total: listaDeUsuarios.length });
    } catch (error) {
        respuesta.status(500).json({ ok: false, error: error.message });
    }
});

// GET /api/usuarios/:id — Obtener por ID
enrutadorUsuarios.get('/:id', async (peticion, respuesta) => {
    try {
        const usuarioEncontrado = await UsuarioModel.buscarPorId(peticion.params.id);
        if (!usuarioEncontrado)
            return respuesta.status(404).json({ ok: false, error: 'Usuario no encontrado' });
        respuesta.json({ ok: true, data: usuarioEncontrado });
    } catch (error) {
        respuesta.status(500).json({ ok: false, error: error.message });
    }
});

// POST /api/usuarios — Crear nuevo usuario
enrutadorUsuarios.post('/', async (peticion, respuesta) => {
    try {
        const { nombre, email, username, password_hash } = peticion.body;
        if (!nombre || !email || !username || !password_hash)
            return respuesta.status(400).json({ ok: false, error: 'Faltan campos requeridos' });

        const resultadoCreacion = await UsuarioModel.crear(peticion.body);
        respuesta.status(201).json({ ok: true, mensaje: 'Usuario creado', data: resultadoCreacion });
    } catch (error) {
        // ORA-00001: constraint única violada (email o username duplicado)
        if (error.message.includes('ORA-00001'))
            return respuesta.status(409).json({ ok: false, error: 'Email o username ya existe' });
        respuesta.status(500).json({ ok: false, error: error.message });
    }
});

// PUT /api/usuarios/:id — Actualizar usuario
enrutadorUsuarios.put('/:id', async (peticion, respuesta) => {
    try {
        const resultadoActualizacion = await UsuarioModel.actualizar(peticion.params.id, peticion.body);
        if (resultadoActualizacion.filasAfectadas === 0)
            return respuesta.status(404).json({ ok: false, error: 'Usuario no encontrado' });
        respuesta.json({ ok: true, mensaje: 'Usuario actualizado' });
    } catch (error) {
        respuesta.status(500).json({ ok: false, error: error.message });
    }
});

// DELETE /api/usuarios/:id — Baja lógica (activo = 'N')
enrutadorUsuarios.delete('/:id', async (peticion, respuesta) => {
    try {
        await UsuarioModel.eliminar(peticion.params.id);
        respuesta.json({ ok: true, mensaje: 'Usuario desactivado' });
    } catch (error) {
        respuesta.status(500).json({ ok: false, error: error.message });
    }
});

module.exports = enrutadorUsuarios;

server.js — Servidor principal

JavaScript — Punto de entrada de la aplicación
require('dotenv').config();  // Cargar variables de .env al inicio

const express  = require('express');
const cors     = require('cors');
const path     = require('path');
const { iniciarPool, cerrarPool } = require('./db/pool');

const aplicacion = express();
const PUERTO     = process.env.PORT || 3000;

// Middlewares globales
aplicacion.use(cors());                              // Habilitar CORS para el frontend
aplicacion.use(express.json());                      // Parsear JSON en peticiones
aplicacion.use(express.static(path.join(__dirname, 'public')));  // Servir HTML/CSS/JS

// Rutas de la API REST
aplicacion.use('/api/usuarios',  require('./routes/usuarios'));
aplicacion.use('/api/productos', require('./routes/productos'));
aplicacion.use('/api/pedidos',   require('./routes/pedidos'));

// Middleware de manejo de errores (siempre al final)
aplicacion.use(require('./middleware/errorHandler'));

// Función que inicia el pool Oracle XE ANTES de aceptar peticiones
async function iniciarServidor() {
    try {
        await iniciarPool();   // Primero el pool, luego el servidor HTTP
        aplicacion.listen(PUERTO, () => {
            console.log(`[SERVIDOR] Activo en http://localhost:${PUERTO}`);
            console.log(`[API] Endpoints en http://localhost:${PUERTO}/api`);
            console.log('[ORACLE] Pool conectado a XEPDB1 via modo Thin');
        });
    } catch (errorDeInicio) {
        console.error('Error al iniciar el servidor:', errorDeInicio.message);
        process.exit(1);
    }
}

// Cierre limpio del pool al detener el servidor (Ctrl+C)
process.on('SIGINT', async () => {
    console.log('\n[SERVIDOR] Cerrando servidor...');
    await cerrarPool();
    process.exit(0);
});

const BASE_API = '/api';   // Prefijo de todos los endpoints

// Cargar y mostrar productos desde la API de Node.js
async function cargarProductos(terminoDeBusqueda = '') {
    try {
        // Si hay texto de búsqueda, agrega el parámetro ?buscar=...
        const urlConsulta = terminoDeBusqueda
            ? `${BASE_API}/productos?buscar=${encodeURIComponent(terminoDeBusqueda)}`
            : `${BASE_API}/productos`;

        // Llamada fetch — comunicación HTTP entre navegador y Node.js
        const respuestaHTTP = await fetch(urlConsulta);
        const datosRecibidos = await respuestaHTTP.json();

        if (!datosRecibidos.ok) throw new Error(datosRecibidos.error);
        renderizarProductos(datosRecibidos.data);

    } catch (errorDeRed) {
        console.error('Error al cargar productos:', errorDeRed.message);
    }
}

// Renderizar las tarjetas de producto en el DOM
function renderizarProductos(listaDeProductos) {
    const contenedor = document.getElementById('contenedorProductos');
    contenedor.innerHTML = listaDeProductos.length === 0
        ? '<p>Sin resultados.</p>'
        : listaDeProductos.map(producto => `
            <div class="tarjeta">
                <h3>${producto.NOMBRE}</h3>
                <small>${producto.CATEGORIA}</small>
                <p class="precio">${Number(producto.PRECIO).toLocaleString('es-PY')} Gs.</p>
                <p class="stock">Stock: ${producto.STOCK}</p>
                <button class="boton" onclick="agregarAlCarrito(${producto.PRODUCTO_ID},'${producto.NOMBRE}',${producto.PRECIO})">
                    + Agregar
                </button>
            </div>`).join('');
}

// Búsqueda con retraso (evita llamar la API en cada tecla)
let timerDeBusqueda;
function buscarConRetraso(textoBuscado) {
    clearTimeout(timerDeBusqueda);
    timerDeBusqueda = setTimeout(() => cargarProductos(textoBuscado), 400);
}

// Carrito de compras en memoria
let carritoDeCompras = [];
function agregarAlCarrito(idProducto, nombreProducto, precioUnitario) {
    const itemExistente = carritoDeCompras.find(i => i.producto_id === idProducto);
    itemExistente
        ? itemExistente.cantidad++
        : carritoDeCompras.push({ producto_id: idProducto, nombre: nombreProducto, precio_unitario: precioUnitario, cantidad: 1 });
    console.log(`Carrito: ${carritoDeCompras.reduce((suma, i) => suma + i.cantidad, 0)} items`);
}

// Enviar el pedido al servidor Node.js que lo guarda en Oracle XE
async function confirmarPedido() {
    if (!carritoDeCompras.length) return;
    try {
        const respuesta = await fetch(`${BASE_API}/pedidos`, {
            method:  'POST',
            headers: { 'Content-Type': 'application/json' },
            body:    JSON.stringify({
                usuario_id: 2,
                items:      carritoDeCompras,
                direccion:  'Asunción, Paraguay'
            })
        });
        const datos = await respuesta.json();
        if (datos.ok) carritoDeCompras = [];
        console.log('Pedido creado:', datos);
    } catch (error) {
        console.error('Error al confirmar pedido:', error.message);
    }
}

// Cargar productos al abrir la página
document.addEventListener('DOMContentLoaded', () => cargarProductos());

Manejo de Errores y Buenas Prácticas

middleware/errorHandler.js — Interceptar errores Oracle XE
// Mapa de errores Oracle XE → respuesta HTTP adecuada
const mapaDeErroresOracle = {
    'ORA-00001': { estado: 409, mensaje: 'Registro duplicado (constraint única)' },
    'ORA-01400': { estado: 400, mensaje: 'Campo obligatorio está nulo' },
    'ORA-02291': { estado: 400, mensaje: 'FK inválida: registro padre no existe' },
    'ORA-02292': { estado: 409, mensaje: 'No se puede eliminar: tiene registros relacionados' },
    'ORA-20001': { estado: 400, mensaje: 'Stock insuficiente para el pedido' },
    'ORA-01017': { estado: 401, mensaje: 'Credenciales Oracle incorrectas' },
    'ORA-12541': { estado: 503, mensaje: 'Oracle XE no disponible (listener)' }
};

function manejarErrores(error, peticion, respuesta, siguiente) {
    // Registrar el error en consola con timestamp
    console.error(`[${new Date().toISOString()}] ${peticion.method} ${peticion.path} →`, error.message);

    // Buscar si el error corresponde a un código Oracle conocido
    for (const [codigoOracle, respuestaPreparada] of Object.entries(mapaDeErroresOracle)) {
        if (error.message.includes(codigoOracle)) {
            return respuesta.status(respuestaPreparada.estado).json({
                ok:     false,
                error:  respuestaPreparada.mensaje,
                codigo: codigoOracle
            });
        }
    }

    // Error genérico no mapeado
    respuesta.status(500).json({ ok: false, error: 'Error interno del servidor' });
}

module.exports = manejarErrores;

Verificar datos en Oracle XE desde DBeaver

SQL — Verificar datos guardados por la API
-- Ver pedidos creados desde la API con el total calculado por el trigger:
SELECT p.pedido_id,
       u.nombre || ' ' || u.apellido AS cliente,
       p.estado,
       TO_CHAR(p.total, 'FM999,999,990 Gs.') AS total,
       TO_CHAR(p.fecha_pedido, 'DD/MM/YYYY HH24:MI:SS') AS fecha
FROM   pedidos p
JOIN   usuarios u ON p.usuario_id = u.usuario_id
ORDER BY p.fecha_pedido DESC;

-- Verificar que el trigger actualizó los totales correctamente:
SELECT ped.pedido_id,
       ped.total                          AS total_calculado_por_trigger,
       SUM(dp.cantidad * dp.precio_unitario) AS total_verificado_manual
FROM   pedidos ped
JOIN   detalle_pedidos dp ON ped.pedido_id = dp.pedido_id
GROUP BY ped.pedido_id, ped.total;

-- Ver logs generados automáticamente por los triggers:
SELECT l.log_id, u.username, l.accion, l.descripcion,
       TO_CHAR(l.fecha_accion, 'DD/MM/YYYY HH24:MI:SS') AS fecha
FROM   logs_acceso l
LEFT JOIN usuarios u ON l.usuario_id = u.usuario_id
ORDER BY l.fecha_accion DESC;
