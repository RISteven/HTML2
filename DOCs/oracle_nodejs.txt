     - Requiere Oracle Instant Client instalado en C:\oracle\instantclient_21_x

// db/connection.js — Conexión simple para pruebas
const oracledb = require('oracledb');

// MODO THIN: no requiere Oracle Client adicional
// Compatible con Oracle XE 21c y la PDB XEPDB1
oracledb.thin = true; // Activar explícitamente modo Thin

const config = {
    user:          'admin_web',
    password:      'AdminWeb2024#',
    connectString: 'localhost:1521/XEPDB1'  // PDB de Oracle XE 21c
};

async function obtenerConexion() {
    try {
        const conexion = await oracledb.getConnection(config);
        console.log('Conexión a Oracle XE (XEPDB1) exitosa');
        return conexion;
    } catch (err) {
        console.error('Error al conectar a Oracle XE:', err.message);
        throw err;
    }
}

async function cerrarConexion(conexion) {
    if (conexion) {
        try {
            await conexion.close();
        } catch (err) {
            console.error('Error al cerrar conexión:', err.message);
        }
    }
}

module.exports = { obtenerConexion, cerrarConexion };

3. POOL DE CONEXIONES (CONEXIÓN EFICIENTE)
================================================================================

-- El Pool mantiene conexiones pre-abiertas para reutilizarlas.
-- Evita la sobrecarga de abrir/cerrar conexión en cada petición HTTP.
-- IMPORTANTE para Oracle XE: poolMax no debe superar ~10 por los límites de la edición.

-----------------------------------------------------------------------
// db/pool.js
const oracledb = require('oracledb');

oracledb.thin = true; // Modo Thin — sin Oracle Client adicional

const poolConfig = {
    user:          process.env.DB_USER     || 'admin_web',
    password:      process.env.DB_PASSWORD || 'AdminWeb2024#',
    connectString: process.env.DB_CONNECT  || 'localhost:1521/XEPDB1',

    // Ajuste para Oracle XE 21c (límite de 2 CPUs y 2GB RAM):
    poolMin:       1,   // Mínimo de conexiones siempre abiertas
    poolMax:       5,   // Máximo de conexiones simultáneas (XE: no superar 10)
    poolIncrement: 1,
    poolTimeout:   60,

    // Retornar filas como objetos JS ({ COLUMNA: valor }) en vez de arrays
    outFormat: oracledb.OUT_FORMAT_OBJECT
};

async function iniciarPool() {
    try {
        await oracledb.createPool(poolConfig);
        console.log(`Pool Oracle XE iniciado: ${poolConfig.connectString}`);
    } catch (err) {
        console.error('Error al crear pool Oracle XE:', err.message);
        throw err;
    }
}

async function cerrarPool() {
    try {
        await oracledb.getPool().close(10);
        console.log('Pool Oracle XE cerrado correctamente');
    } catch (err) {
        console.error('Error al cerrar pool:', err.message);
    }
}

// Ejecutar una query usando el pool (función base reutilizable)
async function ejecutarQuery(sql, params = [], opciones = {}) {
    let conexion;
    try {
        conexion = await oracledb.getConnection();
        const resultado = await conexion.execute(sql, params, {
            outFormat:  oracledb.OUT_FORMAT_OBJECT,
            autoCommit: true,
            ...opciones
        });
        return resultado;
    } catch (err) {
        console.error('Error en query Oracle XE:', err.message);
        throw err;
    } finally {
        if (conexion) await conexion.close();
    }
}

module.exports = { iniciarPool, cerrarPool, ejecutarQuery };

// db/db.js — Métodos CRUD genéricos sobre Oracle XE
const { ejecutarQuery } = require('./pool');
const oracledb = require('oracledb');

const db = {

    // SELECT — Devuelve array de objetos [{COL: val}, ...]
    async query(sql, params = []) {
        const result = await ejecutarQuery(sql, params);
        return result.rows;
    },

    // SELECT — Devuelve solo el primer objeto o null
    async queryOne(sql, params = []) {
        const result = await ejecutarQuery(sql, params);
        return result.rows[0] || null;
    },

    // INSERT / UPDATE / DELETE — Devuelve rowsAffected
    async execute(sql, params = []) {
        const result = await ejecutarQuery(sql, params, { autoCommit: true });
        return {
            rowsAffected: result.rowsAffected,
            outBinds:     result.outBinds
        };
    },

    // INSERT con retorno del ID (usando RETURNING INTO)
    async insertarConId(sql, params = {}) {
        const result = await ejecutarQuery(sql, params, {
            autoCommit: true,
            outFormat:  oracledb.OUT_FORMAT_OBJECT
        });
        return result.outBinds;
    }
};

module.exports = db;

// models/usuarioModel.js
const db = require('../db/db');

const UsuarioModel = {

    async listarTodos() {
        const sql = `
            SELECT u.USUARIO_ID,
                   u.NOMBRE || ' ' || u.APELLIDO AS NOMBRE_COMPLETO,
                   u.EMAIL, u.USERNAME,
                   r.NOMBRE AS ROL, u.ACTIVO,
                   TO_CHAR(u.FECHA_REGISTRO, 'DD/MM/YYYY') AS FECHA_REGISTRO
            FROM   usuarios u
            JOIN   roles r ON u.rol_id = r.rol_id
            ORDER  BY u.nombre
        `;
        return await db.query(sql);
    },

    async buscarPorId(id) {
        const sql = `
            SELECT u.*, r.nombre AS nombre_rol
            FROM   usuarios u
            JOIN   roles r ON u.rol_id = r.rol_id
            WHERE  u.usuario_id = :id
        `;
        return await db.queryOne(sql, [id]);
    },

    async buscarPorEmail(email) {
        const sql = `
            SELECT u.*, r.nombre AS nombre_rol
            FROM   usuarios u
            JOIN   roles r ON u.rol_id = r.rol_id
            WHERE  LOWER(u.email) = LOWER(:email) AND u.activo = 'S'
        `;
        return await db.queryOne(sql, [email]);
    },

    async crear(datos) {
        const sql = `
            INSERT INTO usuarios
                (rol_id, nombre, apellido, email, username, password_hash, telefono)
            VALUES
                (:rol_id, :nombre, :apellido, :email, :username, :password_hash, :telefono)
        `;
        return await db.execute(sql, {
            rol_id:        datos.rol_id,
            nombre:        datos.nombre,
            apellido:      datos.apellido,
            email:         datos.email,
            username:      datos.username,
            password_hash: datos.password_hash,
            telefono:      datos.telefono || null
        });
    },

    async actualizar(id, datos) {
        const sql = `
            UPDATE usuarios
            SET    nombre   = :nombre,
                   apellido = :apellido,
                   telefono = :telefono,
                   activo   = :activo
            WHERE  usuario_id = :id
        `;
        return await db.execute(sql, {
            nombre:   datos.nombre,
            apellido: datos.apellido,
            telefono: datos.telefono || null,
            activo:   datos.activo || 'S',
            id:       id
        });
    },

    async eliminar(id) {
        return await db.execute(
            "UPDATE usuarios SET activo = 'N' WHERE usuario_id = :id",
            [id]
        );
    }
};

module.exports = UsuarioModel;

// models/productoModel.js
const db = require('../db/db');

const ProductoModel = {

    async listarActivos() {
        const sql = `
            SELECT p.PRODUCTO_ID, c.NOMBRE AS CATEGORIA,
                   p.NOMBRE, p.DESCRIPCION,
                   p.PRECIO, p.STOCK, p.IMAGEN_URL
            FROM   productos p
            JOIN   categorias c ON p.categoria_id = c.categoria_id
            WHERE  p.activo = 'S'
            ORDER  BY c.nombre, p.nombre
        `;
        return await db.query(sql);
    },

    async buscarPorId(id) {
        return await db.queryOne(`
            SELECT p.*, c.nombre AS nombre_categoria
            FROM   productos p
            JOIN   categorias c ON p.categoria_id = c.categoria_id
            WHERE  p.producto_id = :id
        `, [id]);
    },

    async buscarPorNombre(nombre) {
        return await db.query(`
            SELECT p.PRODUCTO_ID, p.NOMBRE, p.PRECIO, p.STOCK, c.NOMBRE AS CATEGORIA
            FROM   productos p
            JOIN   categorias c ON p.categoria_id = c.categoria_id
            WHERE  UPPER(p.nombre) LIKE UPPER('%' || :nombre || '%') AND p.activo = 'S'
            ORDER  BY p.nombre
        `, [nombre]);
    },

    async crear(datos) {
        return await db.execute(`
            INSERT INTO productos
                (categoria_id, nombre, descripcion, precio, stock, imagen_url)
            VALUES
                (:categoria_id, :nombre, :descripcion, :precio, :stock, :imagen_url)
        `, {
            categoria_id: datos.categoria_id,
            nombre:       datos.nombre,
            descripcion:  datos.descripcion || null,
            precio:       datos.precio,
            stock:        datos.stock || 0,
            imagen_url:   datos.imagen_url || null
        });
    },

    async actualizar(id, datos) {
        return await db.execute(`
            UPDATE productos
            SET    nombre      = :nombre,
                   descripcion = :descripcion,
                   precio      = :precio,
                   stock       = :stock
            WHERE  producto_id = :id
        `, { ...datos, id });
    },

    async eliminar(id) {
        return await db.execute(
            "UPDATE productos SET activo = 'N' WHERE producto_id = :id",
            [id]
        );
    }
};

// models/pedidoModel.js
const db    = require('../db/db');
const oracledb = require('oracledb');

const PedidoModel = {

    async listarPorUsuario(usuario_id) {
        return await db.query(`
            SELECT ped.PEDIDO_ID, ped.ESTADO,
                   TO_CHAR(ped.TOTAL, 'FM999,999,990') AS TOTAL,
                   TO_CHAR(ped.FECHA_PEDIDO, 'DD/MM/YYYY HH24:MI') AS FECHA,
                   COUNT(dp.detalle_id) AS ITEMS
            FROM   pedidos ped
            LEFT JOIN detalle_pedidos dp ON ped.pedido_id = dp.pedido_id
            WHERE  ped.usuario_id = :usuario_id
            GROUP  BY ped.pedido_id, ped.estado, ped.total, ped.fecha_pedido
            ORDER  BY ped.fecha_pedido DESC
        `, [usuario_id]);
    },

    async obtenerDetalle(pedido_id) {
        return await db.query(`
            SELECT dp.DETALLE_ID, pr.NOMBRE AS PRODUCTO,
                   dp.CANTIDAD,
                   TO_CHAR(dp.PRECIO_UNITARIO, 'FM999,990') AS PRECIO_UNIT,
                   TO_CHAR(dp.SUBTOTAL, 'FM999,990') AS SUBTOTAL
            FROM   detalle_pedidos dp
            JOIN   productos pr ON dp.producto_id = pr.producto_id
            WHERE  dp.pedido_id = :pedido_id
        `, [pedido_id]);
    },

    // Creación de pedido con transacción manual (el trigger actualiza el total)
    async crear(usuario_id, items, direccion) {
        let conexion;
        try {
            conexion = await oracledb.getConnection();

            // Insertar el pedido
            await conexion.execute(
                `INSERT INTO pedidos (usuario_id, estado, direccion_envio)
                 VALUES (:uid, 'PENDIENTE', :dir)`,
                { uid: usuario_id, dir: direccion },
                { autoCommit: false }
            );

            // Obtener el pedido_id recién generado
            const resId = await conexion.execute(
                `SELECT MAX(pedido_id) AS pid FROM pedidos WHERE usuario_id = :uid`,
                [usuario_id],
                { outFormat: oracledb.OUT_FORMAT_OBJECT }
            );
            const pedido_id = resId.rows[0].PID;

            // Insertar cada item (el trigger trg_actualizar_total_pedido
            // actualiza el total del pedido automáticamente)
            for (const item of items) {
                await conexion.execute(
                    `INSERT INTO detalle_pedidos
                         (pedido_id, producto_id, cantidad, precio_unitario)
                     VALUES (:pid, :prod, :cant, :precio)`,
                    {
                        pid:    pedido_id,
                        prod:   item.producto_id,
                        cant:   item.cantidad,
                        precio: item.precio_unitario
                    },
                    { autoCommit: false }
                );

                // Descontar stock (el trigger trg_validar_stock lanza error si stock < 0)
                await conexion.execute(
                    `UPDATE productos SET stock = stock - :cant
                     WHERE producto_id = :pid AND stock >= :cant`,
                    { cant: item.cantidad, pid: item.producto_id },
                    { autoCommit: false }
                );
            }

            await conexion.commit();

            // Obtener el total final calculado por el trigger
            const resPedido = await conexion.execute(
                `SELECT total FROM pedidos WHERE pedido_id = :pid`,
                [pedido_id],
                { outFormat: oracledb.OUT_FORMAT_OBJECT }
            );

            return {
                pedido_id,
                total: resPedido.rows[0].TOTAL,
                items: items.length
            };

        } catch (err) {
            if (conexion) await conexion.rollback();
            throw err;
        } finally {
            if (conexion) await conexion.close();
        }
    },

    async actualizarEstado(pedido_id, estado) {
        const validos = ['PENDIENTE','PROCESANDO','ENVIADO','ENTREGADO','CANCELADO'];
        if (!validos.includes(estado)) throw new Error(`Estado inválido: ${estado}`);
        return await db.execute(
            `UPDATE pedidos SET estado = :estado WHERE pedido_id = :pid`,
            { estado, pid: pedido_id }
        );
    }
};

module.exports = PedidoModel;

// routes/usuarios.js
const express      = require('express');
const router       = express.Router();
const UsuarioModel = require('../models/usuarioModel');

router.get('/', async (req, res) => {
    try {
        const usuarios = await UsuarioModel.listarTodos();
        res.json({ ok: true, data: usuarios, total: usuarios.length });
    } catch (err) {
        res.status(500).json({ ok: false, error: err.message });
    }
});

router.get('/:id', async (req, res) => {
    try {
        const usuario = await UsuarioModel.buscarPorId(req.params.id);
        if (!usuario) return res.status(404).json({ ok: false, error: 'No encontrado' });
        res.json({ ok: true, data: usuario });
    } catch (err) {
        res.status(500).json({ ok: false, error: err.message });
    }
});

router.post('/', async (req, res) => {
    try {
        const { nombre, email, username, password_hash } = req.body;
        if (!nombre || !email || !username || !password_hash) {
            return res.status(400).json({ ok: false, error: 'Faltan campos requeridos' });
        }
        const resultado = await UsuarioModel.crear(req.body);
        res.status(201).json({ ok: true, mensaje: 'Usuario creado', data: resultado });
    } catch (err) {
        if (err.message.includes('ORA-00001'))
            return res.status(409).json({ ok: false, error: 'Email o username ya existe' });
        res.status(500).json({ ok: false, error: err.message });
    }
});

router.put('/:id', async (req, res) => {
    try {
        const resultado = await UsuarioModel.actualizar(req.params.id, req.body);
        if (resultado.rowsAffected === 0)
            return res.status(404).json({ ok: false, error: 'Usuario no encontrado' });
        res.json({ ok: true, mensaje: 'Usuario actualizado' });
    } catch (err) {
        res.status(500).json({ ok: false, error: err.message });
    }
});

router.delete('/:id', async (req, res) => {
    try {
        await UsuarioModel.eliminar(req.params.id);
        res.json({ ok: true, mensaje: 'Usuario desactivado' });
    } catch (err) {
        res.status(500).json({ ok: false, error: err.message });
    }
});

module.exports = router;

        res.status(500).json({ ok: false, error: err.message });
    }
});

router.get('/:id/detalle', async (req, res) => {
    try {
        const detalle = await PedidoModel.obtenerDetalle(req.params.id);
        res.json({ ok: true, data: detalle });
    } catch (err) {
        res.status(500).json({ ok: false, error: err.message });
    }
});

router.post('/', async (req, res) => {
    try {
        const { usuario_id, items, direccion } = req.body;
        if (!usuario_id || !items || items.length === 0)
            return res.status(400).json({ ok: false, error: 'Datos incompletos' });
        const resultado = await PedidoModel.crear(usuario_id, items, direccion);
        res.status(201).json({ ok: true, mensaje: 'Pedido creado', data: resultado });
    } catch (err) {
        // ORA-20001 es lanzado por el trigger trg_validar_stock
        if (err.message.includes('ORA-20001'))
            return res.status(400).json({ ok: false, error: 'Stock insuficiente para el pedido' });
        res.status(500).json({ ok: false, error: err.message });
    }
});

router.patch('/:id/estado', async (req, res) => {
    try {
        await PedidoModel.actualizarEstado(req.params.id, req.body.estado);
        res.json({ ok: true, mensaje: 'Estado actualizado' });
    } catch (err) {
        res.status(400).json({ ok: false, error: err.message });
    }
});

module.exports = router;

// public/js/app.js
const API = '/api';

function mostrarEstado(texto, tipo = 'ok') {
    const el = document.getElementById('estado');
    el.textContent = texto;
    el.className   = tipo;
    setTimeout(() => (el.className = ''), 4000);
}

async function cargarProductos(q = '') {
    try {
        const url  = q ? `${API}/productos?buscar=${encodeURIComponent(q)}` : `${API}/productos`;
        const resp = await fetch(url);
        const data = await resp.json();
        if (!data.ok) throw new Error(data.error);
        renderizar(data.data);
        mostrarEstado(`${data.total} productos desde Oracle XE`);
    } catch (err) {
        mostrarEstado('Error: ' + err.message, 'error');
    }
}

function renderizar(productos) {
    const c = document.getElementById('productos');
    c.innerHTML = productos.length === 0
        ? '<p>Sin resultados.</p>'
        : productos.map(p => `
            <div class="card">
                <h3>${p.NOMBRE}</h3>
                <small>${p.CATEGORIA}</small>
                <p class="precio">${Number(p.PRECIO).toLocaleString('es-PY')} Gs.</p>
                <p class="stock">Stock: ${p.STOCK}</p>
                <button class="btn" onclick="agregar(${p.PRODUCTO_ID},'${p.NOMBRE}',${p.PRECIO})">
                    + Carrito
                </button>
            </div>`).join('');
}

let timerBusqueda;
function buscar(v) {
    clearTimeout(timerBusqueda);
    timerBusqueda = setTimeout(() => cargarProductos(v), 400);
}

let carrito = [];
function agregar(id, nombre, precio) {
    const item = carrito.find(i => i.producto_id === id);
    item ? item.cantidad++ : carrito.push({ producto_id: id, nombre, precio_unitario: precio, cantidad: 1 });
    mostrarEstado(`"${nombre}" agregado. Carrito: ${carrito.reduce((s,i)=>s+i.cantidad,0)} items`);
}

async function crearPedido() {
    if (!carrito.length) return mostrarEstado('Carrito vacío', 'error');
    try {
        const resp = await fetch(`${API}/pedidos`, {
            method:  'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ usuario_id: 2, items: carrito, direccion: 'Asunción, Paraguay' })
        });
        const data = await resp.json();
        if (!data.ok) throw new Error(data.error);
        mostrarEstado(`Pedido #${data.data.pedido_id} creado — Total: ${Number(data.data.total).toLocaleString('es-PY')} Gs.`);
        carrito = [];
    } catch (err) {
        mostrarEstado('Error al crear pedido: ' + err.message, 'error');
    }
}

document.addEventListener('DOMContentLoaded', () => cargarProductos());

// middleware/errorHandler.js
function errorHandler(err, req, res, next) {
    console.error(`[${new Date().toISOString()}] ${req.method} ${req.path} →`, err.message);

    const oraErrors = {
        'ORA-00001': { status: 409, msg: 'Registro duplicado (constraint única)' },
        'ORA-01400': { status: 400, msg: 'Campo obligatorio nulo' },
        'ORA-02291': { status: 400, msg: 'FK inválida: registro padre no existe' },
        'ORA-02292': { status: 409, msg: 'No se puede eliminar: tiene registros relacionados' },
        'ORA-20001': { status: 400, msg: 'Stock insuficiente' },
        'ORA-01017': { status: 401, msg: 'Credenciales Oracle incorrectas' },
        'ORA-12541': { status: 503, msg: 'Oracle XE no disponible (listener)' }
    };

    for (const [codigo, resp] of Object.entries(oraErrors)) {
        if (err.message.includes(codigo))
            return res.status(resp.status).json({ ok: false, error: resp.msg, codigo });
    }

    res.status(500).json({ ok: false, error: 'Error interno del servidor' });
}

module.exports = errorHandler;

-- Errores comunes de Node.js + Oracle XE:

   ORA-01017: invalid username/password
   → Revisar DB_USER y DB_PASSWORD en .env
   → Verificar que el usuario exista en XEPDB1 (no en CDB raíz)

   ORA-12514: TNS:listener does not know of service XEPDB1
   → XEPDB1 no está abierta. Ejecutar:
     sqlplus sys/Oracle21c#@localhost:1521/XE as sysdba
     ALTER PLUGGABLE DATABASE XEPDB1 OPEN;

   NJS-500: DPI-1047: Cannot locate a 64-bit Oracle Client library
   → Asegurarse de que oracledb.thin = true está activo en pool.js
   → O instalar Oracle Instant Client 21c

   NJS-003: invalid connection
   → El pool no se inició antes de procesar peticiones.
   → Verificar que iniciarPool() se llama en iniciarServidor()
